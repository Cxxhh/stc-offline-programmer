# 2.7 STC串口驱动集成设计

## 概述

本文档描述如何将现有的USART2底层实现封装为BSP层设备接口，供ISP协议层使用。

> ✅ **HAL层已完成**：USART2初始化和中断服务已在CubeMX生成代码中实现。
> ⏳ **BSP封装待实现**：需要创建 `bsp_stc_uart.c/h` 封装层。

## 硬件配置

### 引脚分配

| 信号 | 引脚 | GPIO模式 | 说明 |
|------|------|----------|------|
| USART2_TX | PA2 | AF_PP | 发送引脚 |
| USART2_RX | PA3 | AF_PP (上拉) | 接收引脚，防止浮空帧错误 |

### 串口参数（默认配置）

| 参数 | 值 | 说明 |
|------|-----|------|
| 波特率 | 2400 bps | STC握手波特率 |
| 数据位 | 9位 | 8数据+1校验位 |
| 校验位 | 偶校验(EVEN) | STC8/12/15系列要求 |
| 停止位 | 1位 | - |
| 过采样 | 16倍 | 提高接收稳定性 |

### 中断配置

| 中断 | 优先级 | 说明 |
|------|--------|------|
| USART2_IRQn | 0 | 最高优先级，确保不丢数据 |

## 现有代码分析

### usart.c 中的初始化

```c
void MX_USART2_UART_Init(void)
{
    // 时钟配置
    PeriphClkInit.PeriphClockSelection = RCC_PERIPHCLK_USART2;
    PeriphClkInit.Usart2ClockSelection = RCC_USART2CLKSOURCE_PCLK1;
    
    // GPIO配置 (PA2-TX, PA3-RX)
    // RX引脚配置上拉，防止浮空帧错误
    
    // 串口参数
    USART_InitStruct.BaudRate = 2400;
    USART_InitStruct.DataWidth = LL_USART_DATAWIDTH_9B;
    USART_InitStruct.Parity = LL_USART_PARITY_EVEN;
    USART_InitStruct.StopBits = LL_USART_STOPBITS_1;
    
    // 使能接收中断
    LL_USART_EnableIT_RXNE(USART2);
}
```

### stm32g4xx_it.c 中的中断处理

```c
void USART2_IRQHandler(void)
{
    // 1. 错误处理：ORE/FE/NE/PE
    if (LL_USART_IsActiveFlag_ORE(USART2)) { ... }
    if (LL_USART_IsActiveFlag_FE(USART2)) { ... }
    if (LL_USART_IsActiveFlag_NE(USART2)) { ... }
    if (LL_USART_IsActiveFlag_PE(USART2)) { ... }
    
    // 2. 正常接收
    if (LL_USART_IsActiveFlag_RXNE(USART2)) {
        data = LL_USART_ReceiveData8(USART2);
        HandleUart2IT(data);
    }
}
```

## BSP封装设计

### 接口定义

```c
/* BSP/Serial/bsp_stc_uart.h */

#ifndef __BSP_STC_UART_H__
#define __BSP_STC_UART_H__

#include "../Interface/dev_serial.h"
#include "../bsp_common.h"

/* 校验位类型 */
typedef enum {
    STC_UART_PARITY_NONE = 0,  /* 8N1 - STC89/10/11系列 */
    STC_UART_PARITY_EVEN = 1   /* 8E1 - STC8/12/15系列 */
} stc_uart_parity_t;

/* 初始化/反初始化 */
void bsp_stc_uart_init(uint32_t baudrate);
void bsp_stc_uart_deinit(void);

/* 数据收发 */
int bsp_stc_uart_send(const uint8_t *data, uint16_t len);
int bsp_stc_uart_send_nonblock(const uint8_t *data, uint16_t len);
int bsp_stc_uart_read(uint8_t *data, uint16_t max_len, uint16_t *actual_len);

/* 配置修改 */
int bsp_stc_uart_set_baudrate(uint32_t baudrate);
int bsp_stc_uart_set_parity(stc_uart_parity_t parity);

/* 缓冲区管理 */
uint16_t bsp_stc_uart_get_rx_count(void);
void bsp_stc_uart_flush_rx(void);

/* 获取设备实例（符合统一设备接口） */
serial_dev_t *bsp_stc_uart_get_device(void);

/* 中断回调（由stm32g4xx_it.c调用） */
void bsp_stc_uart_rx_callback(uint8_t data);

#endif /* __BSP_STC_UART_H__ */
```

### 关键实现要点

#### 1. 波特率动态切换

```c
int bsp_stc_uart_set_baudrate(uint32_t baudrate)
{
    // 禁用USART
    LL_USART_Disable(USART2);
    
    // 修改波特率
    LL_USART_SetBaudRate(USART2, 
                         LL_RCC_GetUSARTClockFreq(LL_RCC_USART2_CLKSOURCE),
                         LL_USART_PRESCALER_DIV1,
                         LL_USART_OVERSAMPLING_16,
                         baudrate);
    
    // 重新使能USART
    LL_USART_Enable(USART2);
    
    // 等待就绪
    while (!(LL_USART_IsActiveFlag_TEACK(USART2) && 
             LL_USART_IsActiveFlag_REACK(USART2)));
    
    return 0;
}
```

#### 2. 校验位切换

```c
int bsp_stc_uart_set_parity(stc_uart_parity_t parity)
{
    LL_USART_Disable(USART2);
    
    if (parity == STC_UART_PARITY_NONE) {
        // 8N1模式
        LL_USART_SetDataWidth(USART2, LL_USART_DATAWIDTH_8B);
        LL_USART_SetParity(USART2, LL_USART_PARITY_NONE);
    } else {
        // 8E1模式
        LL_USART_SetDataWidth(USART2, LL_USART_DATAWIDTH_9B);
        LL_USART_SetParity(USART2, LL_USART_PARITY_EVEN);
    }
    
    LL_USART_Enable(USART2);
    while (!(LL_USART_IsActiveFlag_TEACK(USART2) && 
             LL_USART_IsActiveFlag_REACK(USART2)));
    
    return 0;
}
```

#### 3. 环形缓冲区集成

```c
#include "../../Service/ringbuffer.h"

static ringbuffer_t s_rx_buffer;
static uint8_t      s_rx_buffer_data[256];

void bsp_stc_uart_init(uint32_t baudrate)
{
    // 初始化环形缓冲区
    ringbuffer_init(&s_rx_buffer, s_rx_buffer_data, sizeof(s_rx_buffer_data));
    
    // 初始化USART2
    MX_USART2_UART_Init();
    
    // 如果需要修改波特率
    if (baudrate != 2400) {
        bsp_stc_uart_set_baudrate(baudrate);
    }
}

/* 中断回调 - 由USART2_IRQHandler调用 */
void bsp_stc_uart_rx_callback(uint8_t data)
{
    ringbuffer_write(&s_rx_buffer, &data, 1);
}
```

## 中断集成修改

需要修改 `Src/stm32g4xx_it.c` 中的 `HandleUart2IT()` 函数，改为调用BSP层回调：

```c
// 修改前（当前实现）
void HandleUart2IT(uint8_t data)
{
    // 直接处理数据...
}

// 修改后（集成BSP层）
#include "../BSP/Serial/bsp_stc_uart.h"

void HandleUart2IT(uint8_t data)
{
    // 调用BSP层回调
    bsp_stc_uart_rx_callback(data);
}
```
